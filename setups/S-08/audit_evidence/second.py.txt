"""Vtoroy i posleduyushchie vizity otdelno ot pervogo. Tot zhe test, ta zhe neytral stenda."""
import numpy as np, pickle, datetime as dt
o=pickle.load(open('work/060/visits.pkl','rb'))
R='data/market/NQ/'
hi=np.load(R+'high.npy'); lo=np.load(R+'low.npy'); ts=np.load(R+'close_ts_utc_ns.npy'); N=len(hi)
BUD=2880; D=[2,3,5]
NEUTRAL={2:46.1,3:47.1,5:48.0}
res={(v,d):[0,0] for v in range(1,5) for d in D}
for x in o:
    p,nu,nr,side,u,line,nv,starts,gaps,fv,cen=x
    if nv<2 or u<=0: continue
    end=min(p+BUD,N-2)
    for vi,stp in enumerate(starts[:4],start=1):
        a=p+stp
        if a+12>=end: break
        h=np.asarray(hi[a+1:end+1],dtype=float); l=np.asarray(lo[a+1:end+1],dtype=float)
        if len(h)<20: break
        for d in D:
            dist=d*u
            aw=np.flatnonzero(h>=line+dist) if side=='north' else np.flatnonzero(l<=line-dist)
            if len(aw)==0: continue
            j=int(aw[0]); h2=h[j+1:]; l2=l[j+1:]
            if len(h2)<2: continue
            if side=='north': bk=np.flatnonzero(l2<=line); fr=np.flatnonzero(h2>=line+2*dist)
            else:             bk=np.flatnonzero(h2>=line); fr=np.flatnonzero(l2<=line-2*dist)
            b=int(bk[0]) if len(bk) else 10**9; f=int(fr[0]) if len(fr) else 10**9
            if b==f==10**9: continue
            res[(vi,d)][0 if b<f else 1]+=1
print("Dolya vozvrata k linii po nomeru vizita, %  (neytral stenda v skobkah)")
print(f"{'vizit':>6} " + " ".join(f"{'d='+str(d)+' ('+str(NEUTRAL[d])+')':>14}" for d in D))
for v in range(1,5):
    row=f"{v:>6} "
    for d in D:
        a,b=res[(v,d)]; n=a+b
        row+=f"{100*a/max(n,1):>7.1f} ({n//1000}k) " if n>=1000 else f"{100*a/max(n,1):>7.1f} ({n})  "
    print(row)
